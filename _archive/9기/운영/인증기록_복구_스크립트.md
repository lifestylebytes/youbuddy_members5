(async () => {
  const FIX = {
    '김옥진': {
      d1:['2026-09-14','19:10'], d2:['2026-09-15','20:59'], d3:['2026-09-16','20:49'],
      d4:['2026-09-17','19:56'], d5:['2026-09-18','17:40'], d6:['2026-09-21','11:15'],
      d7:['2026-09-22','14:32'], d8:['2026-09-23','14:07'], d9:['2026-09-24','15:36'],
      d10:['2026-09-25','22:36'], d11:['2026-09-28','17:38'], d12:['2026-09-29','20:51'],
      d13:['2026-09-30','19:04'], d14:['2026-10-01','22:24'], d15:['2026-10-02','17:07'],
      d16:['2026-10-05','16:23'], d17:['2026-10-06','20:43'], d18:['2026-10-07','21:53'],
      d19:['2026-10-08','21:54'], d20:['2026-10-09','15:00'],
    },
    '황유진': { d20:['2026-10-09','23:03'] },
    '강혜영': { d18:['2026-10-08','13:19'] },   // 예정일 10/7 → 지각 유지
  };
  for (const name in FIX) {
    const st = await window.supabaseRpc('get_member_app_state', { p_cohort:'9기', p_member_key:name });
    st.verified = st.verified || {};
    st.verified_at = st.verified_at || {};
    st.verified_time = st.verified_time || {};
    const added = [];
    for (const k in FIX[name]) {
      if (!st.verified[k]) added.push(k);
      st.verified[k] = true;
      st.verified_at[k] = FIX[name][k][0];
      st.verified_time[k] = FIX[name][k][1];
    }
    await window.supabaseRpc('upsert_member_app_state', {
      p_cohort:'9기', p_member_key:name, p_member_name:name,
      p_tier: st.tier || 'basic', p_app_state: st,
    });
    const chk = await window.supabaseRpc('get_member_app_state', { p_cohort:'9기', p_member_key:name });
    const n = Object.keys(chk.verified || {}).filter(k => chk.verified[k]).length;
    console.log(name, '· 복구', added.join(',') || '(없음)', '→ 현재 인증', n, '일');
  }
  console.log('완료. 기대값 → 김옥진 20 / 황유진 20 / 강혜영 11');
})();

## 원인과 조치 (수정 배포 완료)

`mergeWithServerBeforeSave` 가 서버 상태를 읽어 "더하는 방향으로만" 합치게 돼 있었는데,
그 읽기가 실패했을 때(`loadRemoteState` 의 `catch → return null`) **"서버에 아무것도 없다"** 와
구분되지 않았습니다. 네트워크가 한 번 튀는 순간, 낡은 로컬 사본이 그대로 올라갔습니다.

고친 것 세 가지:

1. **`loadRemoteStateForMerge`** 추가. 읽기 실패와 행 없음을 구분
2. **서버를 못 읽으면 저장 포기** 후 다음 기회에 재시도 (예전엔 원본 snapshot 을 그대로 업로드)
3. **안전망**: 합친 결과의 인증 일수가 서버보다 적으면 서버 값 유지. `final_test` 도 한 번 생기면 유지
